======================================================================
Mutation 1 - Adicionar um livro
======================================================================
POST /graphql
mutation NovoLivro($input: NovoLivroInput!) {
  adicionarLivro(input: $input) {
    id
    titulo
    disponivel
    autor {
      nome
    }
  }
}

Variáveis: {"input": {"titulo": "Laços de Família", "ano": 1960, "genero": "Contos", "autorId": "2"}}

Resposta (HTTP 200):
{
  "data": {
    "adicionarLivro": {
      "id": "6",
      "titulo": "Laços de Família",
      "disponivel": true,
      "autor": {
        "nome": "Clarice Lispector"
      }
    }
  }
}

======================================================================
POST /graphql
mutation {
  adicionarLivro(input: {titulo: "Livro X", ano: 2024, genero: "Teste", autorId: "99"}) {
    id
  }
}

Resposta (HTTP 200):
{
  "errors": [
    {
      "message": "Autor com id 99 não existe",
      "locations": [
        {
          "line": 2,
          "column": 3
        }
      ],
      "path": [
        "adicionarLivro"
      ],
      "extensions": {
        "classification": "BAD_REQUEST"
      }
    }
  ],
  "data": null
}

======================================================================
Introspection - O cliente consulta o próprio Schema (auto-documentação)
======================================================================
POST /graphql
query {
  __type(name: "Livro") {
    name
    description
    fields {
      name
      description
    }
  }
}

Resposta (HTTP 200):
{
  "data": {
    "__type": {
      "name": "Livro",
      "description": "Livro do acervo da biblioteca",
      "fields": [
        {
          "name": "id",
          "description": null
        },
        {
          "name": "titulo",
          "description": null
        },
        {
          "name": "ano",
          "description": "Ano de publicação"
        },
        {
          "name": "genero",
          "description": null
        },
        {
          "name": "nota",
          "description": "Nota média dos leitores, de 0 a 5"
        },
        {
          "name": "disponivel",
          "description": "true se o livro está na estante, false se está emprestado"
        },
        {
          "name": "autor",
          "description": "Autor do livro (relacionamento Livro -> Autor)"
        }
      ]
    }
  }
}
